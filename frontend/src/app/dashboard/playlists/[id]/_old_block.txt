function ItemTransitionEditor({ item, defaultTransition, defaultDurationMs, disabled, pending, onSave }: {
  item: PlaylistItem
  defaultTransition: TransitionName
  defaultDurationMs: number
  disabled: boolean
  pending: boolean
  onSave: (data: { transition: TransitionName | null; transition_ms: number | null }) => void
}) {
  type Selection = TransitionName | 'inherit'
  const [selection, setSelection] = useState<Selection>(item.transition || 'inherit')
  const [durationMs, setDurationMs] = useState(item.transition_ms || defaultDurationMs)
  const effectiveTransition = selection === 'inherit' ? defaultTransition : selection
  const effectiveDuration = selection === 'inherit' ? defaultDurationMs : durationMs
  return (
    <div className="mt-4 border-t border-hairline pt-4">
      <div className="mb-3 flex items-center justify-between"><p className="flex items-center gap-2 text-xs font-semibold text-muted-foreground"><Sparkles className="size-3.5 text-primary" /> Handoff to next ad</p><Badge variant={selection === 'inherit' ? 'secondary' : 'outline'}>{selection === 'inherit' ? 'Playlist default' : 'Custom'}</Badge></div>
      <div className="grid gap-4 sm:grid-cols-[minmax(170px,1fr)_minmax(170px,1fr)_96px] sm:items-end">
        <div><Label className="mb-2 text-xs text-muted-foreground">Transition</Label><Select value={selection} disabled={disabled || pending} onValueChange={(value: Selection | null) => value && setSelection(value)}><SelectTrigger className="w-full"><SelectValue>{(value: Selection | null) => value === 'inherit' ? `Use default (${transitionLabel(defaultTransition)})` : value ? transitionLabel(value) : 'Choose transition'}</SelectValue></SelectTrigger><SelectContent><SelectItem value="inherit">Use playlist default</SelectItem>{transitionOptions.map((option) => <SelectItem key={option.value} value={option.value}>{option.label}</SelectItem>)}</SelectContent></Select></div>
        <div className={cn(selection === 'inherit' && 'opacity-50')}><div className="mb-2 flex items-center justify-between"><Label htmlFor={`transition-duration-${item.id}`} className="text-xs text-muted-foreground">Duration</Label><span className="font-mono text-xs text-muted-foreground">{effectiveDuration} ms</span></div><input id={`transition-duration-${item.id}`} type="range" min={100} max={3000} step={50} value={durationMs} disabled={disabled || pending || selection === 'inherit'} onChange={(event) => setDurationMs(Number(event.target.value))} className="h-10 w-full accent-primary" /></div>
        <TransitionPreview transition={effectiveTransition} durationMs={effectiveDuration} compact />
      </div>
      {!disabled && <div className="mt-3 flex justify-end"><Button size="sm" variant="outline" disabled={pending} onClick={() => onSave(selection === 'inherit' ? { transition: null, transition_ms: null } : { transition: selection, transition_ms: durationMs })}>{pending ? 'Saving…' : <><Check data-icon="inline-start" /> Save handoff</>}</Button></div>}
    </div>
  )
}

function ScheduleEditor({ item, disabled, pending, onSave }: {
  item: PlaylistItem
  disabled: boolean
  pending: boolean
  onSave: (data: { duration: number; start_at: string | null; end_at: string | null; schedule: { days_of_week: number[]; start_time: string | null; end_time: string | null } | null }) => void
}) {
  const [duration, setDuration] = useState(item.duration)
  const [startAt, setStartAt] = useState(dateTimeLocal(item.start_at))
  const [endAt, setEndAt] = useState(dateTimeLocal(item.end_at))
  const [days, setDays] = useState<number[]>(item.schedule?.days_of_week || [])
  const [windowStart, setWindowStart] = useState(item.schedule?.start_time?.slice(0, 5) || '')
  const [windowEnd, setWindowEnd] = useState(item.schedule?.end_time?.slice(0, 5) || '')

  const isScheduled = Boolean(startAt || endAt || days.length || windowStart || windowEnd)
  const save = () => {
    if (startAt && endAt && endAt <= startAt) { toast.error('The end date must be after the start date'); return }
    const hasWindow = Boolean(days.length || windowStart || windowEnd)
    onSave({
      duration,
      start_at: startAt || null,
      end_at: endAt || null,
      schedule: hasWindow ? { days_of_week: days, start_time: windowStart || null, end_time: windowEnd || null } : null,
    })
  }

  return (
    <div className="mt-4 border-t border-hairline pt-4">
      <div className="mb-3 flex items-center justify-between"><p className="flex items-center gap-2 text-xs font-semibold text-muted-foreground"><CalendarClock className="size-3.5 text-primary" /> Playback rules</p><Badge variant={isScheduled ? 'warning' : 'success'}>{isScheduled ? 'Scheduled' : 'Always active'}</Badge></div>
      <fieldset disabled={disabled || pending} className="grid gap-4 sm:grid-cols-2 xl:grid-cols-[110px_1fr_1fr]">
        <div><Label htmlFor={`duration-${item.id}`} className="mb-2 text-xs text-muted-foreground">Duration</Label><div className="relative"><Input id={`duration-${item.id}`} type="number" min={1} max={86400} value={duration} onChange={(event) => setDuration(Math.max(1, Number(event.target.value)))} className="pr-7" /><span className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-xs text-muted-foreground/70">s</span></div></div>
        <div><Label htmlFor={`start-${item.id}`} className="mb-2 text-xs text-muted-foreground">Starts</Label><Input id={`start-${item.id}`} type="datetime-local" value={startAt} onChange={(event) => setStartAt(event.target.value)} /></div>
        <div><Label htmlFor={`end-${item.id}`} className="mb-2 text-xs text-muted-foreground">Expires</Label><Input id={`end-${item.id}`} type="datetime-local" value={endAt} onChange={(event) => setEndAt(event.target.value)} /></div>
      </fieldset>
      <fieldset disabled={disabled || pending} className="mt-4 grid gap-4 xl:grid-cols-[1fr_150px_150px]">
        <div><legend className="mb-2 text-xs font-medium text-muted-foreground">Days of week <span className="font-normal text-muted-foreground/70">(none = every day)</span></legend><div className="flex gap-1.5">{dayLabels.map((label, day) => <button key={day} type="button" aria-label={dayNames[day]} aria-pressed={days.includes(day)} onClick={() => setDays((current) => current.includes(day) ? current.filter((value) => value !== day) : [...current, day].sort())} className={cn('grid size-9 place-items-center rounded-lg border text-xs font-bold transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring', days.includes(day) ? 'border-primary bg-primary text-primary-foreground' : 'border-hairline bg-card text-muted-foreground hover:border-border')}>{label}</button>)}</div></div>
        <div><Label htmlFor={`from-${item.id}`} className="mb-2 text-xs text-muted-foreground">Daily from</Label><Input id={`from-${item.id}`} type="time" value={windowStart} onChange={(event) => setWindowStart(event.target.value)} /></div>
        <div><Label htmlFor={`until-${item.id}`} className="mb-2 text-xs text-muted-foreground">Daily until</Label><Input id={`until-${item.id}`} type="time" value={windowEnd} onChange={(event) => setWindowEnd(event.target.value)} /></div>
      </fieldset>
      {!disabled && <div className="mt-4 flex justify-end"><Button size="sm" variant="outline" onClick={save} disabled={pending}>{pending ? 'Saving…' : <><Check data-icon="inline-start" /> Save rules</>}</Button></div>}
    </div>
  )
}

function SortableItemCard({ item, defaultTransition, defaultDurationMs, canEdit, saving, onRemove, onSave, onSaveTransition }: {
  item: PlaylistItem
  defaultTransition: TransitionName
  defaultDurationMs: number
  canEdit: boolean
  saving: boolean
  onRemove: () => void
  onSave: Parameters<typeof ScheduleEditor>[0]['onSave']
  onSaveTransition: Parameters<typeof ItemTransitionEditor>[0]['onSave']
}) {
  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id: item.id, disabled: !canEdit })
  return (
    <Card ref={setNodeRef} style={{ transform: CSS.Transform.toString(transform), transition }} className={cn('border-0 bg-card py-0 ring-1 ring-hairline', isDragging ? 'z-20 opacity-80 shadow-2xl' : 'shadow-[0_1px_2px_rgba(15,23,42,.04)]')}>
      <CardContent className="p-4 sm:p-5">
        <div className="flex items-center gap-3">
          <button type="button" {...attributes} {...listeners} disabled={!canEdit} className="grid size-9 shrink-0 cursor-grab place-items-center rounded-lg text-muted-foreground/40 hover:bg-muted hover:text-muted-foreground active:cursor-grabbing disabled:cursor-default focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring" aria-label={`Move ${item.content.name}`}><GripVertical className="size-5" /></button>
          <MediaThumbnail item={item.content} className="size-16 shrink-0 rounded-xl" />
          <div className="min-w-0 flex-1"><p className="truncate font-semibold text-foreground">{item.content.name}</p><p className="mt-1 flex items-center gap-1 text-xs capitalize text-muted-foreground/70"><Clock3 className="size-3" /> {item.duration}s · {item.content.type}</p></div>
          {canEdit && <Button variant="ghost" size="icon" className="text-muted-foreground/40 hover:bg-destructive/10 hover:text-destructive" onClick={onRemove} aria-label={`Remove ${item.content.name}`}><Trash2 /></Button>}
        </div>
        <ItemTransitionEditor key={`${item.transition || 'inherit'}-${item.transition_ms || defaultDurationMs}`} item={item} defaultTransition={defaultTransition} defaultDurationMs={defaultDurationMs} disabled={!canEdit} pending={saving} onSave={onSaveTransition} />
        <ScheduleEditor item={item} disabled={!canEdit} pending={saving} onSave={onSave} />
      </CardContent>
    </Card>
  )
}
